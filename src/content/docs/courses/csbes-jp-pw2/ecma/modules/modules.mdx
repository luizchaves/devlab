## Resumo dos Cenários

export const singleCJS = `
module.exports = sum
const sum = require('./lib')
`;

export const multiCJS = `
module.exports = { sum, minus }
const { sum, minus } = require('./lib')
`;

export const singleESM = `
export default sum
import sum from './lib'
`;

export const multiESM = `
export { sum, minus }
import { sum, minus } from './lib'
`;
